#include<bits/stdc++.h>
using namespace std;
 
void solve(){
    double discount, currentprice;
    cin >> discount >> currentprice;
    
    double originalprice = currentprice / (1 - discount / 100);
    cout << fixed << setprecision(2) << originalprice << endl;
}
 
int main()
{
    int t = 1;
    //cin >> t;
    while(t--) solve();
}
