#include<bits/stdc++.h>
using namespace std;
int main()
{
    long long a,b;
    cin>>a>>b;
    cout<<"floor "<<a<<" / "<<b<<" = "<<floor(a*1.00/b)<<endl;
    cout<<"ceil "<<a<<" / "<<b<<" = "<<ceil(a*1.00/b)<<endl;
    cout<<"round "<<a<<" / "<<b<<" = "<<round(a*1.00/b)<<endl;
}
